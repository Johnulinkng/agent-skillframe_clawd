1.2 推荐架构:意图-动作模式
这是AI驱动的Web3产品(如Fini、DODO X、Matcha)使用的标准模式:
Plain Text
sequenceDiagram
    participant 用户
    participant 前端
    participant AI Agent后端
    participant 区块链
    
    用户->>前端: 自然语言输入
    前端->>AI Agent后端: API调用
    
    Note over AI Agent后端: LLM推理<br/>工具执行
    
    AI Agent后端->>前端: 返回带client_action的响应
    
    Note over 前端: 解析client_action<br/>显示确认UI
    
    用户->>前端: 确认/拒绝
    前端->>区块链: 钱包签名
核心是AI Agent返回一个意图(它希望前端做什么),而不是执行本身。
1.3 client_action协议定义
每个AI Agent响应可能包含零个或多个client_action对象。前端必须从响应中解析这些对象并渲染适当的UI。
TypeScript
// 定义在: src/types/client-action.ts

interface AgentResponse {
    message: string;           // 文本响应(始终存在)
    client_actions?: ClientAction[];  // 可选的动作指令
}

interface ClientAction {
    type: string;              // 动作类型枚举(见下表)
    params: Record<string, any>;  // 特定于动作的参数
    display?: {                // 可选的显示提示
        label?: string;        // 按钮标签
        style?: 'primary' | 'warning' | 'danger';
        icon?: string;
    };
}

interface AgentResponse {
    message: string;
    client_actions?: ClientAction[];
    sessionId: string;
}

// 前端处理器
function handleAgentResponse(response: AgentResponse) {
    // 1. 显示消息
    displayMessage(response.message);
    // 2. 处理动作
    if (response.client_actions) {
        response.client_actions.forEach(action => {
            handleClientAction(action);
        });
    }
}
3.3 client_action处理
TypeScript
function handleClientAction(action: ClientAction) {
    switch (action.type) {
        case 'OPEN_TRADE_WINDOW':
            // 打开交易窗口,预填充参数
            openTradeModal({
                symbol: action.params.symbol,
                side: action.params.side,
                tradeType: action.params.tradeType,
                amount: action.params.amount,
                price: action.params.price,
            });
            break;

        case 'CONFIRM_TRANSACTION':
            // 显示交易确认对话框
            showConfirmDialog({
                title: '确认交易',
                description: action.params.description,
                onConfirm: async () => {
                    // 使用Privy签名并广播
                    const tx = await privy.signTransaction({
                        to: action.params.to,
                        value: action.params.value,
                        data: action.params.data,
                    });
                    await broadcastTransaction(tx);
                },
                onCancel: () => {
                    sendMessage('用户取消了交易。');
                },
            });
            break;

        case 'NAVIGATE':
            router.push(action.params.path);
            break;

        case 'OPEN_DEPOSIT':
            router.push('/deposit');
            break;

        case 'QUICK_ACTIONS':
            // 在聊天中渲染动作按钮
            renderActionButtons(action.params.buttons);
            break;

        case 'SHOW_TABLE':
            // 在聊天消息区域渲染表格
            renderTable(action.params);
            break;
    }
}
